{# ---------- 步骤 2：签名机制 ---------- #}

{% trans "签名机制（HMAC-SHA256）" %}

{% trans "需要签名（auth）的接口，每次请求都要带上公共参数：" %} app_id / timestamp / nonce / sign{% trans "。" %}

  1. {% trans "取" %}{% trans "除 sign 外" %}{% trans "的全部参数（含 app_id/timestamp/nonce 与业务参数），过滤空值；" %}
  2. {% trans "按键名 ASCII 升序排序；" %}
  3. {% trans "拼接为" %} key=value&key=value…{% trans "；" %}
  4. {% trans "以 APPSECRET 为密钥做" %} HMAC-SHA256{% trans "，输出小写 hex 即 sign。" %}

{% trans "timestamp 为秒级时间戳，有效窗口 ±5 分钟；nonce 每次请求唯一（服务端去重防重放）。" %}

{% trans "Python 签名示例" %} 
import hashlib
import hmac
import time
import secrets

APP_ID = 'app_yourAppId'
APP_SECRET = 'sk_yourAppSecret'


def build_sign(params: dict, secret: str) -> str:
    """{% trans "按键名 ASCII 升序拼接 key=value，HMAC-SHA256 输出小写 hex" %}"""
    items = sorted(
        (k, str(v)) for k, v in params.items()
        if k != 'sign' and v not in (None, '')
    )
    raw = '&'.join(f'{k}={v}' for k, v in items)
    return hmac.new(secret.encode(), raw.encode(), hashlib.sha256).hexdigest()


def signed_params(biz: dict):
    params = dict(biz)
    params.update({
        'app_id': APP_ID,
        'timestamp': str(int(time.time())),
        'nonce': secrets.token_hex(8),
    })
    params['sign'] = build_sign(params, APP_SECRET)
    return params
