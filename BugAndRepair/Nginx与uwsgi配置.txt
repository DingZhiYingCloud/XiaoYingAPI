# ===================== 静态资源配置开始 =====================
# 【请把下面这段 map 放在 nginx 的 http {} 块内（server 块外）】
# S-02/S-11：/media/ 若由 nginx 直服，Django 侧的 _media_serve 不生效，
# 需在此按扩展名强制附件下载，避免上传的危险文件被浏览器内联渲染（存储型 XSS）。
# 扩展名与 XiaoYingAPI/urls.py 的 DANGEROUS_MEDIA_EXTS 保持一致。
map $uri $media_disposition {
    default                                        "";
    ~*\.(svg|html?|xhtml|xml|js|mht|mhtml|php|phtml)$  "attachment";
}

location /static/ {
    alias {XIAOYING_API_ROOT}/static/;
    expires 30d; # 浏览器缓存30天
    add_header Cache-Control "public, no-transform";
    access_log off; # 关闭静态文件日志，减少磁盘IO
}
location /media/ {
    alias {XIAOYING_API_ROOT}/media/;
    expires 30d; # 浏览器缓存30天
    add_header Cache-Control "public, no-transform";
    # S-02/S-11 安全：禁止嗅探 MIME 类型；可渲染/可执行类型强制下载（值为空时 nginx 不发送该头）
    add_header X-Content-Type-Options nosniff;
    add_header Content-Disposition $media_disposition;
    access_log off; # 关闭静态文件日志，减少磁盘IO
}

# ===================== 静态资源配置结束 =====================




# ===================== UWSGI配置开始 =====================
static-map = /static={XIAOYING_API_ROOT}/static
env = LANG=C.UTF-8

# ===================== UWSGI配置结束 =====================




# ===================== SSE 长连接配置开始 =====================
# 【放置位置与上面两段不同】这段不是贴进宝塔面板的站点配置，而是作为一个独立文件放到
# 站点的「扩展目录」里（宝塔重生成站点配置时不会覆盖该目录）：
#     /www/server/panel/vhost/nginx/extension/小影API/console_sse.conf
# 站点配置里已有 `include /www/server/panel/vhost/nginx/extension/小影API/*.conf;`
#
# 用途：海角自动注册的 SSE 进度流，每一步（经代理取注册验证码 / 超级鹰打码）之间有
# 一段无数据下发的时间，而 nginx 的 proxy_read_timeout 默认只有 60s —— 任一步超过
# 60s，nginx 会直接断开连接，浏览器 EventSource 只触发 onerror，页面只能显示
# 「连接中断」，服务端算出来的真实失败原因再也推不过来。
# 实测：线上经 51代理 取码需 62s（代理 IP 从美服连不上，3 次重试各 20s），正好被默认
# 的 60s 掐断。故对该路径单独放宽到 300s，其余 location 仍用 nginx 默认值。
#
# 该段与源码保持一致：{XIAOYING_API_ROOT} 即项目根目录，上游固定为 uwsgi 的 http 监听。
#
location ^~ /console/haijiao/register/stream/ {
    proxy_pass http://127.0.0.1:10005;
    proxy_set_header Host $http_host;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Real-Port $remote_port;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;
    proxy_set_header X-Forwarded-Host $host;
    proxy_set_header X-Forwarded-Port $server_port;
    proxy_set_header Remote-Host $remote_addr;
    proxy_read_timeout 300s;
    proxy_send_timeout 300s;
}

# ===================== SSE 长连接配置结束 =====================
